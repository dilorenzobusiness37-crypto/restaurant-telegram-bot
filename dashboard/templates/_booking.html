{% endmacro %} {% macro register(bookings, today_iso, back='/', live=False, highlight_id=None) %} {% set now = now_hm() %} {% set active = bookings | selectattr('status', 'eq', 'confirmed') | list %} {% set cancelled = bookings | rejectattr('status', 'eq', 'confirmed') | list %} {% set ns = namespace(now_drawn=not live) %} {% if active %}

  2. {{ group.grouper }}
    {% for b in group.list %}{{ arrival(b, today_iso, back, highlight_id) }}{% endfor %}
